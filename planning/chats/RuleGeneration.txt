## ✅ Core Types

### `RulePipeline.cs`

```csharp
public class RulePipeline
{
    private readonly IRuleParser _parser;
    private readonly IRuleIRGenerator _irGenerator;
    private readonly IRuleCompiler _compiler;

    public RulePipeline(IRuleParser parser, IRuleIRGenerator irGen, IRuleCompiler compiler)
    {
        _parser = parser;
        _irGenerator = irGen;
        _compiler = compiler;
    }

    public void Compile(string ruleSetNameOrPath)
    {
        var sources = _parser.ParseRules(ruleSetNameOrPath, LoadRaw(ruleSetNameOrPath));
        var ir = _irGenerator.GenerateIR(sources);
        _compiler.Compile(ir);
    }

    private string LoadRaw(string name)
    {
        // could be a file load, embedded text, or ID lookup
        return File.ReadAllText($"{name}.rule");
    }
}
```

## ✅ Compiler with Fluent Builder API

### `DllRuleCompiler.cs`

```csharp
public class DllRuleCompiler : IRuleCompiler
{
    private string? _outputPath;
    private string? _fileName;
    private string? _targetFramework;
    private string? _namespace;
    private bool _emitDll = false;

    public DllRuleCompiler WithOutputPath(string path)
    {
        _outputPath = path;
        return this;
    }

    public DllRuleCompiler WithFileName(string name)
    {
        _fileName = name;
        return this;
    }

    public DllRuleCompiler WithTargetFramework(string framework)
    {
        _targetFramework = framework;
        return this;
    }

    public DllRuleCompiler WithNamespace(string ns)
    {
        _namespace = ns;
        return this;
    }

    public DllRuleCompiler EmitDll(bool enable)
    {
        _emitDll = enable;
        return this;
    }

    public void Compile(IEnumerable<RuleModel> rules)
    {
        // Step 1: Generate .cs file
        var code = RuleCsGenerator.Emit(rules, _namespace ?? "Generated");

        var filePath = Path.Combine(_outputPath ?? ".", _fileName ?? "GeneratedRules.cs");
        File.WriteAllText(filePath, code);

        if (_emitDll)
        {
            // Step 2: Use Roslyn to emit DLL
            DllBuilder.Compile(filePath, _outputPath!, _targetFramework ?? "net8.0");
        }
    }
}
```

| Element                     | Purpose                           | Notes                                                    |
| --------------------------- | --------------------------------- | -------------------------------------------------------- |
| `RulePrototype`             | Author DSL input                  | Holds `Name`, `Condition`, `Action` delegates            |
| `[RuleGroup("Name")]`       | Declares a group                  | Class-level tag for organization                         |
| `[RuleTemplate]`            | Marks rule for generation         | Field must return a `RulePrototype`                      |
| `[StandingRule(typeof(T))]` | Adds `!ctx.Has<T>()` condition    | Can appear at class or rule level                        |

| Generated rules             | Implement `IRule<T>`              | With `Name`, `CanExecute`, and `Execute` methods         |
| Group registry class        | Appended as `ClassName_Generated` | Exposes `.GetRules()` for easy inclusion in a rulebook   |
